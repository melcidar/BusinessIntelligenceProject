SELECT
  { [Measures].[Exceed Days (any)] } ON COLUMNS,
  ORDER(
    NONEMPTY(
      [dim_city].[Geo].[city_name].Members,
      [Measures].[Exceed Days (any)]
    ),
    [Measures].[Exceed Days (any)],
    DESC
  ) ON ROWS
FROM [AirQ Cube]
WHERE (
  [dim_timemonth].[Time].[year_num].&[2024],
  [dim_param].[Param].[param_name].&[PM10]
)
