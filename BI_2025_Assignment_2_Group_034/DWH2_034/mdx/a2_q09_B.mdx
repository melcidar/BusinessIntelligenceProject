WITH SET Top10 AS
  TOPCOUNT(
    NONEMPTY(
      [dim_city].[Geo].[country_name].Members,
      [Measures].[Reading Events]
    ),
    10,
    [Measures].[Reading Events]
  )
SELECT
  {
    [dim_timemonth].[Time].[quarter_num].&[1],
    [dim_timemonth].[Time].[quarter_num].&[2],
    [dim_timemonth].[Time].[quarter_num].&[3],
    [dim_timemonth].[Time].[quarter_num].&[4]
  } * { [Measures].[Reading Events] } ON COLUMNS,
  Top10 ON ROWS
FROM [AirQ Cube]
WHERE ([dim_timemonth].[Time].[year_num].&[2024]);
